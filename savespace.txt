function rand_basic_1936_RSA(){
    R_m_target = rand_Int(3,30)*500; //Zielentfernung zwischn 1500 und 22500
    RSA_target = rand_Int(1,15); //Ziel RSA zwischen
    goleft = rand_Bool();
    if (goleft) {
        RSA_target = RSA_target * -1;
    }
    BR_target = (RSA_target / R_m_target) * 1936;
    BR = BR_target.toFixed(1);

    K_e = rand_Int(1,36)*10;
    v_e = rand_Int(3,8);
    B_m = rand_Int(0,140);
    B_mleft = rand_Bool();
    if (B_mleft){
        B_m = B_m *-1;
    }

    B_mn = K_e + B_m;
    if (B_mn < 0){
        B_mn = B_mn + 360;
    }
    else if (B_mn > 360){
        B_mn = B_mn - 360;
    }

    OSA = v_e * uboot_sinus(B_mn - K_e);

    TSA_target = RSA_target - OSA;

    v_dmin = parseInt(TSA_target.toFixed(0));
    if (v_dmin < 0){
        v_dmin = v_dmin *-1 +1;
    }
    else {
        v_dmin = v_dmin + 1;
    }

    v_d = rand_Int(v_dmin,25);


    I_m = Math.asin(TSA_target/v_d);
    I_m = (I_m * 360)/(2 * Math.PI)
    if (I_m > 180){
        I_m = I_m - 360
    }
    else if (I_m < -180){
        I_m = I_m + 360
    }
    I_m = I_m /10;
    I_m = I_m.toFixed(0);
    I_m = I_m * 10;

    result = [
        K_e,
        v_e,
        B_mn,
        I_m,
        v_d,
        BR,
        R_m_target,
    ];

    return result;


}

function rand_basic_1936_KA(){
    val = rand_basic_1936_RSA();
    K_e1 = val[0];
    v_e1 = val[1];
    B_mn = val[2];
    I_m = val[3];
    v_d = val[4];
    BR1 = val[5];
    R_m_target = val[6];

    DOSA_target = rand_Int(20,80)/10;

    OSA1 = v_e1 * uboot_sinus(B_mn - K_e1);
    OSA2 = DOSA_target + OSA1;

    DB = (1936/R_m_target) * DOSA_target;

    BR2 = DB + parseFloat(BR1);


    v_dmin = parseInt(OSA2.toFixed(0));
    if (v_dmin<0){
        v_dmin = v_dmin *-1 +1 ;
    }
    if (v_dmin <3){
        v_dmin = 3;
    }
    else {
        v_dmin = v_dmin + 1;
    }
    v_e2 = rand_Int(v_dmin, v_dmin+4);
    B_m2 = Math.asin(OSA2 / v_e2);
    B_m2 = (B_m2 * 360)/(2 * Math.PI);
    K_e2 = B_mn - B_m2;
    K_e2 = K_e2/10;
    K_e2 = K_e2.toFixed(0)*10;
    if (K_e2 < 0){
        K_e2 = K_e2 + 360;
    }
    K_e2 = parseInt(K_e2.toFixed(0));
    BR2 = parseFloat(BR2.toFixed(1));





    result = [
        K_e1,
        v_e1,
        B_mn,
        K_e2,
        v_e2,
        BR1,
        BR2,
        v_d
    ];

    return result;


}
